cmake_minimum_required(VERSION 3.12)
project(cf_dummy LANGUAGES C)
find_package(zstd CONFIG REQUIRED)

set(targets libzstd libzstd_shared)
if(ZSTD_TEST_STATIC)
  list(APPEND targets libzstd_static)
elseif(TARGET zstd::libzstd_static)
  message(FATAL_ERROR "The shared-only package must not export a missing static library")
endif()

get_target_property(default_library zstd::libzstd INTERFACE_LINK_LIBRARIES)
if(NOT default_library STREQUAL "zstd::libzstd_shared")
  message(FATAL_ERROR "The default Zstd target must keep selecting the shared library")
endif()

enable_testing()
foreach(target IN LISTS targets)
  add_executable(test_${target} test.c)
  target_link_libraries(test_${target} PRIVATE zstd::${target})
  add_test(NAME ${target} COMMAND test_${target})
endforeach()
